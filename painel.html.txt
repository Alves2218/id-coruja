    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Painel de Boas-Vindas</title>
    <link rel="stylesheet" href="style.css">
</head>
<body>
    <div class="card-container">
        <h2>Bem-vindo(a)!</h2>
        <p id="welcomeMessage" style="text-align: center; margin: 20px 0; color: #555;">
            Carregando usuário...
        </p>
        <button onclick="logout()" class="btn btn-sair">Sair</button>
    </div>

    <script>
        // Executa assim que a página carrega
        document.addEventListener("DOMContentLoaded", () => {
            const usuarioLogado = localStorage.getItem("usuarioNome");
            const msgElemento = document.getElementById("welcomeMessage");

            if (usuarioLogado) {
                msgElemento.textContent = `Olá, ${usuarioLogado}! Que bom ter você de volta.`;
            } else {
                // Se tentar acessar direto sem logar, volta para o login
                alert("Por favor, faça login primeiro.");
                window.location.href = "index.html"; 
            }
        });

        // Função do botão Sair
        function logout() {
            localStorage.removeItem("usuarioNome"); // Apaga os dados
            window.location.href = "index.html"; // Volta para o login
        }
    </script>
</body>
</html>
