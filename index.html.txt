                <button type="submit" class="btn" style="background: linear-gradient(135deg, #28a745 0%, #20c997 100%); box-shadow: 0 4px 14px rgba(40, 167, 69, 0.3);">Solicitar Cadastro</button>

    <script>
        if (localStorage.getItem('usuarioSessaoAtiva')) {
            window.location.href = 'dashboard.html';
        }

        // Garante que a conta inicial do admin mestre exista com status aprovado
        document.addEventListener("DOMContentLoaded", () => {
            const usuarios = obterBancoUsuarios();
            if (!usuarios['admin']) {
                usuarios['admin'] = { senha: 'admin', aprovado: true };
                localStorage.setItem('banco_usuarios', JSON.stringify(usuarios));
            }
        });

        function toggleInputPassword(inputId, botao) {
            const inputCampo = document.getElementById(inputId);
            inputCampo.type = inputCampo.type === "password" ? "text" : "password";
            botao.textContent = inputCampo.type === "password" ? "Mostrar" : "Ocultar";
        }

        function alternarTelas(event) {
            if(event) event.preventDefault();
            const login = document.getElementById('loginSection');
            const cadastro = document.getElementById('cadastroSection');
            
            document.getElementById('loginPass').type = "password";
            document.getElementById('cadPass').type = "password";
            document.querySelectorAll('.btn-toggle-input').forEach(btn => btn.textContent = "Mostrar");

            if (login.style.display === 'none') {
                login.style.display = 'block';
                cadastro.style.display = 'none';
                document.getElementById('cadastroForm').reset();
            } else {
                login.style.display = 'none';
                cadastro.style.display = 'block';
                document.getElementById('loginForm').reset();
            }
        }

        function obterBancoUsuarios() {
            const banco = localStorage.getItem('banco_usuarios');
            return banco ? JSON.parse(banco) : {};
        }

        function esqueciSenha(event) {
            if(event) event.preventDefault();
            const usuario = prompt("Digite o nome do usuário que deseja recuperar:");
            if (!usuario) return;

            const usuariosCadastrados = obterBancoUsuarios();
            const userKey = usuario.trim();

            if (usuariosCadastrados[userKey]) {
                const novaSenha = prompt(`Usuário localizado! Digite a nova senha para "${userKey}":`);
                if (novaSenha && novaSenha.trim().length >= 4) {
                    usuariosCadastrados[userKey].senha = novaSenha.trim();
                    localStorage.setItem('banco_usuarios', JSON.stringify(usuariosCadastrados));
                    alert("Senha alterada com sucesso!");
                } else if (novaSenha) {
                    alert("A senha precisa ter no mínimo 4 caracteres.");
                }
            } else {
                alert("Usuário não encontrado!");
            }
        }

        // CADASTRO COM SISTEMA DE VALIDAÇÃO REQUERIDA
        document.getElementById('cadastroForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const usuario = document.getElementById('cadUser').value.trim();
            const senha = document.getElementById('cadPass').value.trim();

            if (usuario.length === 0 || senha.length < 4) {
                alert('O usuário não pode ser vazio e a senha precisa ter ao menos 4 caracteres.');
                return;
            }

            const usuariosCadastrados = obterBancoUsuarios();

            if (usuariosCadastrados[usuario]) {
                alert('Este nome de usuário já está sendo usado!');
                return;
            }

            // Salva a nova conta com o status falso de aprovação
            usuariosCadastrados[usuario] = {
                senha: senha,
                aprovado: false
            };
            
            localStorage.setItem('banco_usuarios', JSON.stringify(usuariosCadastrados));

            alert('Solicitação de cadastro enviada! Aguarde a aprovação do administrador para conseguir acessar.');
            alternarTelas();
        });

        // LOGIN COM VERIFICAÇÃO DE STATUS ATIVO
        document.getElementById('loginForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const usuario = document.getElementById('loginUser').value.trim();
            const senha = document.getElementById('loginPass').value.trim();
            const usuariosCadastrados = obterBancoUsuarios();

            const conta = usuariosCadastrados[usuario];

            if (conta && (conta.senha === senha || (typeof conta === 'string' && conta === senha))) {
                // Se for uma conta antiga convertida ou uma conta nova, trata o objeto
                const isAprovado = (typeof conta === 'object') ? conta.aprovado : true;

                if (!isAprovado) {
                    alert('Acesso Negado! Sua conta ainda não foi ativada pelo administrador.');
                    return;
                }

                localStorage.setItem('usuarioSessaoAtiva', usuario);
                window.location.href = 'dashboard.html';
            } else {
                alert('Usuário ou senha incorretos!');
            }
        });
